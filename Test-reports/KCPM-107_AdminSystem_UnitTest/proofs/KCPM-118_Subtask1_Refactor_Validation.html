1. Guard Clause Validation tại GET /dashboard/{warehouseId}:
@GetMapping("/dashboard/{warehouseId}")
@PreAuthorize("hasRole('ADMIN')")
public ResponseEntity<ApiResponse<DashboardStatsResponse>> getDashboardStats(@PathVariable String warehouseId) {
    if (warehouseId == null || warehouseId.trim().isEmpty()) {
        throw new BadRequestException("Mã kho (warehouseId) không được để trống");
    }
    DashboardStatsResponse stats = systemManagementService.getDashboardStats(warehouseId);
    return ResponseEntity.ok(ApiResponse.success(stats, "Dashboard statistics retrieved"));
}
2. Guard Clause Validation tại PUT /config (Khóa tham số rỗng/chuỗi trắng):
@PutMapping("/config")
@PreAuthorize("hasRole('ADMIN')")
public ResponseEntity<ApiResponse<SystemConfig>> updateConfig(
        @RequestParam String key,
        @RequestParam String value,
        @RequestParam String adminId) {

    if (key == null || key.trim().isEmpty()) {
        throw new BadRequestException("Khóa cấu hình (key) không được để trống");
    }
    if (adminId == null || adminId.trim().isEmpty()) {
        throw new BadRequestException("Mã quản trị viên (adminId) không được để trống");
    }

    SystemConfig config = systemManagementService.updateConfig(key, value);
    systemManagementService.logAction(adminId, "UPDATE_CONFIG", "Config " + key + " updated to " + value, "SYSTEM");
    return ResponseEntity.ok(ApiResponse.success(config, "System configuration updated"));
}
